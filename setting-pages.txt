'use client';
import { useAuth } from '@/context/AuthContext';
import { useState, useEffect } from 'react';
import {
  Settings, Code2, Wrench, Download, Trash2, Save, Eye, EyeOff,
  Crown, ShieldCheck, Lock, CheckCircle2, User, Calendar, Zap,
} from 'lucide-react';

const APP_VERSION = '1.0.0';

export default function SettingsPage() {
  const { user, customerRegistrationEnabled, toggleCustomerRegistration, updateOwnerPassword, getOwnerPassword } = useAuth();
  const isOwner = user?.role === 'Owner';

  // Developer States
  const [newPass, setNewPass] = useState('');
  const [showPass, setShowPass] = useState(false);
  const [currentPass, setCurrentPass] = useState('');

  // Tools States
  const [apiUrl, setApiUrl] = useState('https://www.tikwm.com/api/');
  const [quality, setQuality] = useState('hd');
  const [removeWatermark, setRemoveWatermark] = useState(true);
  const [debugMode, setDebugMode] = useState(false);

  const [activeTab, setActiveTab] = useState<'dev' | 'tools' | 'downloader'>('downloader');

  // Feedback toast (pengganti alert() biar terasa premium)
  const [toast, setToast] = useState<string | null>(null);
  const flash = (msg: string) => {
    setToast(msg);
    setTimeout(() => setToast(null), 2200);
  };

  useEffect(() => {
    setCurrentPass(getOwnerPassword());
    const savedApi = localStorage.getItem('tt_api_url');
    const savedQuality = localStorage.getItem('tt_quality');
    const savedWm = localStorage.getItem('tt_wm');
    const savedDebug = localStorage.getItem('tt_debug');
    if (savedApi) setApiUrl(savedApi);
    if (savedQuality) setQuality(savedQuality);
    if (savedWm) setRemoveWatermark(savedWm === 'true');
    if (savedDebug) setDebugMode(savedDebug === 'true');
  }, [getOwnerPassword]);

  const handleSaveDev = () => {
    if (newPass.length < 4) return flash('Password minimal 4 karakter');
    updateOwnerPassword(newPass);
    setCurrentPass(newPass);
    setNewPass('');
    flash('Password Owner berhasil di-update!');
  };

  const handleSaveTools = () => {
    localStorage.setItem('tt_api_url', apiUrl);
    localStorage.setItem('tt_quality', quality);
    localStorage.setItem('tt_wm', String(removeWatermark));
    localStorage.setItem('tt_debug', String(debugMode));
    flash('Konfigurasi berhasil disimpan!');
  };

  const handleToggleRegistration = () => {
    toggleCustomerRegistration();
    flash(customerRegistrationEnabled ? 'Registrasi customer dinonaktifkan' : 'Registrasi customer diaktifkan');
  };

  const handleClearAll = () => {
    if (confirm('Yakin hapus semua data customer & cache?')) {
      localStorage.removeItem('tt_customers');
      localStorage.removeItem('tt_user');
      flash('Data berhasil dibersihkan');
    }
  };

  return (
    <div className="min-h-screen bg-black text-white p-6 relative overflow-hidden">
      {/* Ambient VIP glow background */}
      <div className="pointer-events-none absolute -top-40 -left-40 w-96 h-96 bg-amber-500/10 rounded-full blur-3xl" />
      <div className="pointer-events-none absolute -bottom-40 -right-40 w-96 h-96 bg-amber-400/10 rounded-full blur-3xl" />

      <div className="max-w-4xl mx-auto relative">
        <div className="flex items-center justify-between mb-2">
          <h1 className="text-3xl font-bold flex items-center gap-3">
            <Settings className="w-8 h-8" /> Settings
          </h1>
          {isOwner && (
            <span className="flex items-center gap-1.5 text-xs font-bold px-3 py-1.5 rounded-full bg-gradient-to-r from-amber-300 via-yellow-400 to-amber-500 text-black shadow-lg shadow-amber-500/20">
              <Crown className="w-3.5 h-3.5" /> OWNER VIP
            </span>
          )}
        </div>
        <p className="text-neutral-500 text-sm mb-8">Khusus Developer & Konfigurasi Downloader</p>

        {/* Tabs */}
        <div className="flex gap-2 p-1 bg-neutral-900 border border-neutral-800 rounded-2xl w-fit mb-6">
          <button onClick={() => setActiveTab('downloader')} className={`px-5 py-2.5 rounded-xl text-sm font-semibold flex items-center gap-2 transition-all ${activeTab === 'downloader' ? 'bg-white text-black' : 'text-neutral-400 hover:text-white'}`}>
            <Download className="w-4 h-4" /> Downloader
          </button>
          <button onClick={() => setActiveTab('tools')} className={`px-5 py-2.5 rounded-xl text-sm font-semibold flex items-center gap-2 transition-all ${activeTab === 'tools' ? 'bg-white text-black' : 'text-neutral-400 hover:text-white'}`}>
            <Wrench className="w-4 h-4" /> Tools
          </button>
          <button onClick={() => setActiveTab('dev')} className={`px-5 py-2.5 rounded-xl text-sm font-semibold flex items-center gap-2 transition-all ${activeTab === 'dev' ? 'bg-gradient-to-r from-amber-300 to-amber-500 text-black' : 'text-neutral-400 hover:text-white'}`}>
            <Code2 className="w-4 h-4" /> Developer
            {!isOwner && <Lock className="w-3 h-3" />}
          </button>
        </div>

        {/* CONTENT */}
        <div className="bg-neutral-900/60 border border-neutral-800 rounded-3xl p-6 md:p-8">

          {activeTab === 'downloader' && (
            <div className="space-y-6">
              <h2 className="font-semibold text-lg">Konfigurasi Downloader</h2>
              <div className="grid md:grid-cols-2 gap-4">
                <div>
                  <label className="text-xs text-neutral-400">API Endpoint</label>
                  <input value={apiUrl} onChange={e => setApiUrl(e.target.value)} className="w-full mt-1.5 bg-neutral-800 border border-neutral-700 rounded-xl px-4 py-3 text-sm focus:outline-none focus:border-amber-400/60" />
                </div>
                <div>
                  <label className="text-xs text-neutral-400">Kualitas Video</label>
                  <select value={quality} onChange={e => setQuality(e.target.value)} className="w-full mt-1.5 bg-neutral-800 border border-neutral-700 rounded-xl px-4 py-3 text-sm focus:outline-none focus:border-amber-400/60">
                    <option value="hd">HD - 1080p (No Watermark)</option>
                    <option value="sd">SD - 720p</option>
                    <option value="watermark">With Watermark</option>
                  </select>
                </div>
              </div>
              <div className="flex items-center justify-between bg-neutral-800/80 p-4 rounded-xl border border-neutral-700/50">
                <div><p className="text-sm font-medium">Hapus Watermark Otomatis</p><p className="text-xs text-neutral-500">Pakai server tikwm / ssstik</p></div>
                <button
                  type="button"
                  role="switch"
                  aria-checked={removeWatermark}
                  onClick={() => setRemoveWatermark(v => !v)}
                  className={`w-11 h-6 rounded-full transition-all relative ${removeWatermark ? 'bg-amber-400' : 'bg-neutral-700'}`}
                >
                  <div className={`w-5 h-5 bg-black rounded-full transition-all absolute top-0.5 ${removeWatermark ? 'translate-x-5' : 'translate-x-1'}`} />
                </button>
              </div>
              <button onClick={handleSaveTools} className="w-full py-3 bg-white text-black font-semibold rounded-xl text-sm hover:bg-neutral-200 transition-all">
                <Save className="w-4 h-4 inline mr-2" />Simpan Config Downloader
              </button>
            </div>
          )}

          {activeTab === 'tools' && (
            <div className="space-y-6">
              <h2 className="font-semibold text-lg">Tools & Maintenance</h2>
              <div className="grid gap-3">
                <button
                  onClick={handleToggleRegistration}
                  className="p-4 rounded-xl bg-neutral-800 border border-neutral-700 flex justify-between items-center hover:border-neutral-600 transition-all text-left"
                >
                  <span className="text-sm">Customer Registration</span>
                  <span className={`text-xs px-3 py-1 rounded-full font-semibold ${customerRegistrationEnabled ? 'bg-green-500/20 text-green-400' : 'bg-red-500/20 text-red-400'}`}>
                    {customerRegistrationEnabled ? 'ON' : 'OFF'}
                  </span>
                </button>
                <div className="p-4 rounded-xl bg-neutral-800 border border-neutral-700 flex justify-between items-center">
                  <span className="text-sm">Debug Mode</span>
                  <input type="checkbox" checked={debugMode} onChange={e => setDebugMode(e.target.checked)} className="w-4 h-4 accent-amber-400" />
                </div>
                <button onClick={handleClearAll} className="flex items-center justify-center gap-2 w-full py-3 bg-red-500/10 border border-red-500/20 text-red-400 rounded-xl text-sm hover:bg-red-500/20 transition-all">
                  <Trash2 className="w-4 h-4" /> Clear All Cache & Customer List
                </button>
              </div>
              <button onClick={handleSaveTools} className="w-full py-3 bg-white text-black font-semibold rounded-xl text-sm hover:bg-neutral-200 transition-all">Simpan Tools</button>
            </div>
          )}

          {activeTab === 'dev' && (
            <div className="space-y-6">
              {!isOwner ? (
                <div className="flex flex-col items-center text-center gap-3 py-10">
                  <Lock className="w-10 h-10 text-neutral-600" />
                  <p className="text-sm text-neutral-400 max-w-xs">Halaman ini khusus Owner. Login sebagai Owner untuk mengakses & mengubah pengaturan developer.</p>
                </div>
              ) : (
                <>
                  {/* VIP Profile Card */}
                  <div className="relative overflow-hidden rounded-2xl p-6 border border-amber-400/30 bg-gradient-to-br from-amber-500/10 via-neutral-900 to-neutral-900">
                    <div className="absolute top-0 right-0 w-32 h-32 bg-amber-400/10 rounded-full blur-2xl" />
                    <div className="flex items-center gap-4 relative">
                      <div className="w-16 h-16 rounded-full bg-gradient-to-br from-amber-300 to-amber-600 flex items-center justify-center text-black font-bold text-xl shadow-lg shadow-amber-500/20 flex-shrink-0">
                        {user?.username?.charAt(0)?.toUpperCase() || <User className="w-6 h-6" />}
                      </div>
                      <div className="min-w-0">
                        <div className="flex items-center gap-2 flex-wrap">
                          <p className="font-bold text-lg truncate">{user?.username}</p>
                          <span className="flex items-center gap-1 text-[10px] font-bold px-2 py-0.5 rounded-full bg-gradient-to-r from-amber-300 to-amber-500 text-black">
                            <Crown className="w-3 h-3" /> VIP OWNER
                          </span>
                        </div>
                        <p className="text-xs text-neutral-400 flex items-center gap-1.5 mt-1">
                          <ShieldCheck className="w-3.5 h-3.5 text-amber-400" /> Akses penuh developer aktif
                        </p>
                      </div>
                    </div>
                  </div>

                  <h2 className="font-semibold text-lg">Developer Access</h2>
                  <div>
                    <label className="text-xs text-neutral-400">Current Owner Password</label>
                    <div className="flex gap-2 mt-1.5">
                      <div className="flex-1 relative">
                        <input type={showPass ? 'text' : 'password'} value={currentPass} readOnly className="w-full bg-neutral-800 border border-neutral-700 rounded-xl px-4 py-3 text-sm text-neutral-400" />
                        <button type="button" onClick={() => setShowPass(!showPass)} className="absolute right-3 top-3.5 text-neutral-400 hover:text-white">
                          {showPass ? <EyeOff className="w-4 h-4" /> : <Eye className="w-4 h-4" />}
                        </button>
                      </div>
                    </div>
                  </div>
                  <div>
                    <label className="text-xs text-neutral-400">Ganti Password Baru</label>
                    <input type="password" value={newPass} onChange={e => setNewPass(e.target.value)} placeholder="Password baru..." className="w-full mt-1.5 bg-neutral-800 border border-amber-400/30 focus:border-amber-400 rounded-xl px-4 py-3 text-sm outline-none" />
                  </div>
                  <div className="grid grid-cols-2 gap-3 text-xs">
                    <div className="p-3 bg-neutral-800 rounded-xl flex items-center gap-2">
                      <Zap className="w-3.5 h-3.5 text-amber-400" /> App v{APP_VERSION}
                    </div>
                    <div className="p-3 bg-neutral-800 rounded-xl flex items-center gap-2">
                      <Calendar className="w-3.5 h-3.5 text-amber-400" /> {user?.username} ({user?.role})
                    </div>
                  </div>
                  <button onClick={handleSaveDev} className="w-full py-3 bg-gradient-to-r from-amber-300 to-amber-500 text-black font-bold rounded-xl text-sm hover:brightness-95 transition-all">
                    Update Password Owner
                  </button>
                </>
              )}
            </div>
          )}

        </div>
      </div>

      {/* Toast feedback */}
      {toast && (
        <div className="fixed bottom-6 left-1/2 -translate-x-1/2 flex items-center gap-2 bg-neutral-900 border border-amber-400/30 text-sm px-4 py-3 rounded-xl shadow-2xl z-50">
          <CheckCircle2 className="w-4 h-4 text-amber-400" /> {toast}
        </div>
      )}
    </div>
  );
}
